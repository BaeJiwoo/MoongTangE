function(sandbox_test name source)
  add_executable(${name} ${source})
  target_link_libraries(${name} PRIVATE Common Threads::Threads ${ARGN})
  add_test(NAME ${name} COMMAND ${name})
  set_tests_properties(${name} PROPERTIES TIMEOUT 30)
endfunction()
sandbox_test(MathTests MathTests.cpp)
sandbox_test(PhysicsTests PhysicsTests.cpp)
sandbox_test(WorldTests WorldTests.cpp)
sandbox_test(ServerTests ServerTests.cpp ServerCore)
sandbox_test(LocalTests LocalTests.cpp ServerCore)
sandbox_test(ProtocolTests ProtocolTests.cpp)
sandbox_test(CliTests CliTests.cpp)
