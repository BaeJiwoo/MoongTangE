cmake_minimum_required(VERSION 3.25)
project(PortablePhysicsSandbox VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(SANDBOX_BUILD_CLIENT "Build raylib GUI client" ON)
option(SANDBOX_BUILD_NETWORK "Build Asio TCP transport and server" ON)
include(CTest)
find_package(Threads REQUIRED)
add_library(Common INTERFACE)
target_include_directories(Common INTERFACE "${PROJECT_SOURCE_DIR}")
target_compile_features(Common INTERFACE cxx_std_20)
add_library(ServerCore Server/Core/ServerHost.cpp)
target_link_libraries(ServerCore PUBLIC Common Threads::Threads)
include(cmake/Dependencies.cmake)
if(SANDBOX_BUILD_NETWORK)
  add_library(TcpTransport Server/Network/AsioServer.cpp Client/Transport/TcpClientTransport.cpp)
  target_link_libraries(TcpTransport PUBLIC ServerCore AsioStandalone)
  add_executable(DedicatedServer Server/main.cpp)
  target_link_libraries(DedicatedServer PRIVATE TcpTransport)
endif()
if(SANDBOX_BUILD_CLIENT)
  add_executable(Client Client/main.cpp)
  target_link_libraries(Client PRIVATE ServerCore raylib)
  if(SANDBOX_BUILD_NETWORK)
    target_compile_definitions(Client PRIVATE SANDBOX_HAS_NETWORK)
    target_link_libraries(Client PRIVATE TcpTransport)
  endif()
endif()
if(BUILD_TESTING)
  add_subdirectory(Tests)
endif()
