cmake_minimum_required(VERSION 3.25)
if(CMAKE_HOST_SYSTEM_NAME STREQUAL "Darwin" OR CMAKE_SYSTEM_NAME STREQUAL "Darwin")
  message(FATAL_ERROR "MoongTangE does not support macOS; use Windows or Linux.")
endif()
project(MoongTangE VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(MSVC)
  add_compile_options(/utf-8)
endif()
option(MOONGTANGE_BUILD_CLIENT "Build raylib GUI client" ON)
option(MOONGTANGE_BUILD_NETWORK "Build Standalone Asio TCP transport and server" ON)
include(CTest)
find_package(Threads REQUIRED)
add_library(Common INTERFACE)
target_include_directories(Common INTERFACE "${PROJECT_SOURCE_DIR}")
target_compile_features(Common INTERFACE cxx_std_20)
if(MSVC)
  target_compile_options(Common INTERFACE /W4 /permissive-)
else()
  target_compile_options(Common INTERFACE -Wall -Wextra -Wpedantic)
endif()
add_library(ClientDisplaySettings Client/Rendering/DisplaySettings.cpp
  Client/Rendering/DisplaySettingsStore.cpp)
target_link_libraries(ClientDisplaySettings PUBLIC Common)
if(MSVC)
  target_compile_definitions(ClientDisplaySettings PRIVATE _CRT_SECURE_NO_WARNINGS)
endif()
add_library(ServerCore Server/Core/ServerHost.cpp Server/Core/TickScheduler.cpp
  Server/Core/GameRoom.cpp Server/Core/GameRoomManager.cpp Server/Core/GameRoomOptions.cpp)
target_link_libraries(ServerCore PUBLIC Common Threads::Threads)
add_library(ServerOptions Server/ServerOptions.cpp)
target_link_libraries(ServerOptions PUBLIC Common)
include(cmake/Dependencies.cmake)
if(MOONGTANGE_BUILD_NETWORK)
  include(cmake/StandaloneAsioBackend.cmake)
  include(cmake/Protobuf.cmake)
  add_library(TcpTransport Network/StandaloneAsioBackend.cpp
    Server/Network/StandaloneAsioServer.cpp
    Server/Network/ServerNetworkRuntime.cpp Network/TcpChannel.cpp
    Client/Transport/TcpClientTransport.cpp)
  target_link_libraries(TcpTransport PUBLIC ServerCore StandaloneAsio NetworkProtocol)
  moongtange_configure_standalone_asio_backend(TcpTransport)
  add_library(ServerRuntime Server/ServerApplication.cpp
    Server/Api/MatchmakingControlClient.cpp)
  target_link_libraries(ServerRuntime PUBLIC TcpTransport ServerOptions PRIVATE MatchmakingHttp)
  if(MSVC)
    target_compile_definitions(ServerRuntime PRIVATE _CRT_SECURE_NO_WARNINGS)
  endif()
  add_executable(DedicatedServer Server/main.cpp)
  target_link_libraries(DedicatedServer PRIVATE ServerRuntime)
endif()
if(MOONGTANGE_BUILD_CLIENT)
  add_executable(Client Client/main.cpp Client/ClientApplication.cpp Client/Rendering/Scene.cpp)
  target_link_libraries(Client PRIVATE ClientDisplaySettings ServerCore raylib)
  add_custom_target(moongtange_client_assets ALL
    COMMAND ${CMAKE_COMMAND} -E copy_directory
      "${PROJECT_SOURCE_DIR}/Assets" "$<TARGET_FILE_DIR:Client>/Assets"
    COMMENT "Copying client assets")
  add_dependencies(Client moongtange_client_assets)
  if(MSVC)
    # std::getenv is used read-only for an optional smoke-test screenshot path.
    target_compile_definitions(Client PRIVATE _CRT_SECURE_NO_WARNINGS)
  endif()
  if(MOONGTANGE_BUILD_NETWORK)
    target_compile_definitions(Client PRIVATE MOONGTANGE_HAS_NETWORK)
    target_link_libraries(Client PRIVATE TcpTransport)
  endif()
endif()
if(BUILD_TESTING)
  include(cmake/HeaderChecks.cmake)
  add_subdirectory(Tests)
endif()
